  const manyTasks = tasks.length > 4;
  const imageGridClass =
    images.length === 1
      ? 'max-w-xl'
      : images.length === 3
        ? 'grid-cols-1 sm:grid-cols-2 lg:grid-cols-3'
        : 'grid-cols-1 sm:grid-cols-2';

  return (
    // Blocks are grouped by meaning: title + paragraph read as one unit (tight), then each
    // following block (tasks, images, quote) is set apart by a larger, regular step.
    <div className="space-y-9 sm:space-y-11 lg:space-y-12">
      {title || description ? (
        <div className="space-y-4 sm:space-y-5">
          {title ? (
            <h3
              className="max-w-3xl text-balance text-[1.75rem] font-semibold leading-[1.15] tracking-tight sm:text-4xl lg:text-[2.5rem]"
              style={{ color: presentation.titleColor } satisfies CSSProperties}
            >
              {title}
            </h3>
          ) : null}

          {description ? (
            // ~62–68 characters per line at every size: past that the eye loses its place.
            <p
              className="max-w-[40rem] text-pretty text-base leading-[1.7] sm:text-lg"
              style={{ color: presentation.descriptionColor }}
            >
              {description}
            </p>
          ) : null}
        </div>
      ) : null}

      {presentation.showTasks && tasks.length > 0 ? (
        <ul
          className={`space-y-3 ${
            manyTasks ? 'max-w-3xl md:grid md:grid-cols-2 md:gap-x-10 md:gap-y-3 md:space-y-0' : 'max-w-2xl'
          }`}
        >
          {tasks.map((task, index) => (
            <li
              key={`${index}-${task}`}
              className="flex gap-3 text-[15px] leading-relaxed sm:text-base"
              style={{ color: presentation.taskColor }}
            >
              <PortfolioListMarker
                style={marker.style}
                color={marker.color}
                index={index}
                size={marker.size}
                sizePx={marker.sizePx}
                weight={marker.weight}
                weightAmount={marker.weightAmount}
              />
              <span className="min-w-0">{task}</span>
            </li>
          ))}
        </ul>
      ) : null}

      {images.length > 0 ? (
        <div className={`grid gap-3 sm:gap-4 ${imageGridClass}`}>
          {images.map((src) => (
            // eslint-disable-next-line @next/next/no-img-element
            <img
              key={src}
              {...mediaImageResponsive(src, ABOUT_MEDIA_WIDTHS)}
              sizes={
                images.length > 1
                  ? '(min-width: 1024px) 30vw, (min-width: 640px) 45vw, 90vw'
                  : ABOUT_MEDIA_SIZES
              }
              alt=""
              loading="lazy"
              // Several images stacked on a phone would fill two screens at 4:3: flatter there.
              className={`w-full rounded-2xl object-cover ${
                images.length > 1 ? 'aspect-[16/10] sm:aspect-[4/3]' : 'aspect-[4/3]'
              }`}
            />
          ))}
        </div>
      ) : null}

      <AboutUsQuoteFounder
        quote={quote}
        founderName={founderName}
        founderFunction={founderFunction}
        founderLogo={founderLogo}
        presentation={presentation}
        large
      />
    </div>
  );
}
